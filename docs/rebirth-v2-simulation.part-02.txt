class SmartFilter {
  constructor(host, key, label, selected, onChange) {
    this.host = host;
    this.key = key;
    this.label = label;
    this.selected = selected;
    this.onChange = onChange;
    this.options = optionsForFilter(key);
    host.classList.add("field", "smart");
    host.innerHTML = `<label>${escapeHtml(label)}</label>
      <input autocomplete="off" placeholder="Search ${escapeHtml(label.toLowerCase())}">
      <div class="chips"></div><div class="smart-menu" hidden></div>`;
    this.input = host.querySelector("input");
    this.chips = host.querySelector(".chips");
    this.menu = host.querySelector(".smart-menu");
    this.input.addEventListener("focus", () => this.renderMenu());
    this.input.addEventListener("input", () => this.renderMenu());
    this.render();
  }
  matches() {
    const query = this.input.value.trim().toLowerCase();
    return this.options.filter(value => !this.selected.has(value) && (!query || value.toLowerCase().includes(query))).slice(0, 100);
  }
  renderMenu() {
    const values = this.matches();
    this.menu.innerHTML = values.slice(0, 18).map(value => `<button class="smart-option" data-value="${escapeHtml(value)}"><span>${escapeHtml(value)}</span><span>＋</span></button>`).join("")
      + `<div class="smart-foot">Showing ${Math.min(18, values.length)} of at most 100 matches · ${this.options.length} available</div>`;
    this.menu.hidden = false;
    this.menu.querySelectorAll("[data-value]").forEach(button => button.addEventListener("click", () => {
      this.selected.add(button.dataset.value);
      this.input.value = "";
      this.menu.hidden = true;
      this.render();
      this.onChange();
    }));
  }
  render() {
    this.chips.innerHTML = [...this.selected].map(value => `<span class="chip">${escapeHtml(value)}<button data-remove="${escapeHtml(value)}">×</button></span>`).join("");
    this.chips.querySelectorAll("[data-remove]").forEach(button => button.addEventListener("click", () => {
      this.selected.delete(button.dataset.remove);
      this.render();
      this.onChange();
    }));
  }
}
