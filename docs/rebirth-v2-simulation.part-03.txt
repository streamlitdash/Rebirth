function aggregate(rows) {
  const result = {
    risk: sum(rows.map(row => row.risk)),
    drisk: sum(rows.map(row => row.drisk)),
    pl: sum(rows.map(row => row.pl)),
    open: mean(rows.map(row => row.open)),
    current: mean(rows.map(row => row.current))
  };
  result.move = result.current != null && result.open != null ? result.current - result.open : null;
  return result;
}

/* ---------- promotion ---------- */
const baselinePromotionActivities = new Set(["Rates", "Credit", "FX"]);
function promotionKey(row) {
  return `${row.riskType}|${row.riskGreek}|${row.reportedUnderlying}`;
}
function calculatePromotion(rows) {
  const grouped = new Map();
  rows.filter(row => row.mapped).forEach(row => {
    const key = promotionKey(row);
    if (!grouped.has(key)) grouped.set(key, {riskType: row.riskType, riskGreek: row.riskGreek, underlying: row.reportedUnderlying, risk: 0, drisk: 0, pl: 0});
    const item = grouped.get(key);
    item.risk += row.risk;
    item.drisk += row.drisk;
    item.pl += row.pl;
  });
  const result = new Map();
  grouped.forEach((item, key) => {
    const threshold = thresholds.get(`${item.riskType}|${item.riskGreek}`) || {risk: 2e7, drisk: 5e6, pl: 1e6};
    const ratios = [
      Math.abs(item.risk) / threshold.risk,
      Math.abs(item.drisk) / threshold.drisk,
      Math.abs(item.pl) / threshold.pl
    ];
    const reasons = [];
    if (ratios[0] >= 1) reasons.push("Big Risk");
    if (ratios[1] >= 1) reasons.push("Big dRisk");
    if (ratios[2] >= 1) reasons.push("Big PL");
    result.set(key, {...item, score: Math.max(...ratios), reason: reasons.join(", "), bucket: reasons.length ? item.underlying : "Other"});
  });
  return result;
}
const baselinePromotion = calculatePromotion(riskRows.filter(row => baselinePromotionActivities.has(row.activity)));

}

function renderAggregate(hostId, virtualId, rows, dimension, windowStart, setWindow, openSet) {
  const values = unique(rows.map(row => String(row[dimension]))).sort();
  const windowSize = 8;
  const start = clamp(windowStart, 0, Math.max(0, values.length - windowSize));
  const visible = values.slice(start, start + windowSize);
  const virtual = document.getElementById(virtualId);
  if (values.length > windowSize) {
    virtual.innerHTML = `<div class="virtual-toolbar"><label>Column window <input type="range" min="0" max="${values.length - windowSize}" value="${start}"></label><span class="virtual-label">${start + 1}–${Math.min(start + windowSize, values.length)} of ${values.length} ${dimension}</span></div>`;
    virtual.querySelector("input").addEventListener("input", event => setWindow(+event.target.value));
  } else {
    virtual.innerHTML = "";
  }
  const riskTypes = ["IR", "Credit", "FX", "Commo"].filter(type => rows.some(row => row.riskType === type));
  const records = [{label: "TOTAL", scope: rows, total: true}];
  riskTypes.forEach(riskType => {
    const scope = rows.filter(row => row.riskType === riskType);
    records.push({label: riskType, scope, riskType, expandable: true, open: openSet.has(riskType)});
    if (openSet.has(riskType)) {
      unique(scope.map(row => row.riskGreek)).sort().forEach(greek => records.push({label: greek, scope: scope.filter(row => row.riskGreek === greek), child: true}));
    }
  });
  document.getElementById(hostId).innerHTML = `<table><thead><tr><th class="index-head">Risk / Greek</th>${visible.map(value => `<th>${escapeHtml(value)}</th>`).join("")}<th class="total-col">Total</th></tr></thead>
    <tbody>${records.map(record => {
      const cells = visible.map(value => sum(record.scope.filter(row => String(row[dimension]) === value).map(row => row.pl)));
      const total = sum(record.scope.map(row => row.pl));
      return `<tr class="${record.total ? "total-row" : record.riskType ? "major-row" : ""}">
        <th class="index-cell" style="padding-left:${record.child ? 50 : 10}px">${record.expandable ? `<button class="row-toggle" data-aggregate-toggle="${record.riskType}">${record.open ? "−" : "▸"}</button>` : ""}${escapeHtml(record.label)}</th>
        ${cells.map(value => `<td class="${signClass(value)}">${formatNumber(value, "pl")}</td>`).join("")}
        <td class="total-col ${signClass(total)}">${formatNumber(total, "pl")}</td>
      </tr>`;
    }).join("")}</tbody></table>`;
  document.querySelectorAll(`#${hostId} [data-aggregate-toggle]`).forEach(button => button.addEventListener("click", () => {
    const riskType = button.dataset.aggregateToggle;
    openSet.has(riskType) ? openSet.delete(riskType) : openSet.add(riskType);
    hostId.startsWith("risk") ? renderRiskAggregate() : renderPnlAggregate();
  }));
}

function marketValue(identity, dateIndex, swapIndex = 0, optionIndex = 0) {
  const seed = identity.split("").reduce((total, character) => total + character.charCodeAt(0), 0);
  const base = identity.startsWith("FX") ? 1.1 : identity.startsWith("IR") ? 2.6 : 70;
  return base
    + 0.18 * Math.sin(dateIndex / 6 + swapIndex / 2 + seed / 31)
    + 0.11 * Math.cos(dateIndex / 9 + optionIndex / 2 + seed / 47)
    + 0.32 * Math.exp(-Math.pow(dateIndex - 58, 2) / 45 - Math.pow(swapIndex - 4, 2) / 3 - Math.pow(optionIndex - 2, 2) / 2);
}

function riskHierarchy(rows) {
  const groups = ["riskGreek"];
  if (state.risk.promotionEnabled) groups.push("displayBucket");
  if (state.risk.regionEnabled) groups.push("region");
  groups.push("reportedUnderlying", "tenorSwap", "tenorOption", state.risk.dimension);
  const records = [{label: "TOTAL", depth: 0, scope: rows, total: true, path: ""}];
  function walk(scope, depth, pathParts) {
    if (depth >= groups.length) return;
    const field = groups[depth];
    const grouped = new Map();
    scope.forEach(row => {
      const value = field === "displayBucket" ? displayBucket(row) : String(row[field] ?? "Unspecified");
      if (!grouped.has(value)) grouped.set(value, []);
      grouped.get(value).push(row);
    });
    [...grouped.entries()].sort((a, b) => Math.abs(aggregate(b[1]).pl) - Math.abs(aggregate(a[1]).pl)).slice(0, 24).forEach(([value, child]) => {
      const path = [...pathParts, `${field}=${value}`].join("¦");
      const expandable = depth + 1 < groups.length;
      const open = state.risk.open.has(path);
      records.push({label: value, depth: depth + 1, scope: child, path, expandable, open});
      if (expandable && open) walk(child, depth + 1, [...pathParts, `${field}=${value}`]);
    });
  }
  walk(rows, 0, []);
  return records;
}
